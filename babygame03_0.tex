    \createpin{exec0}{57.5,-147}
	\begin{absnode}[-1]{exec0}{115}{Program Loading [System V ABI]}
		\begin{enumerate}[leftmargin=*,label=(\arabic*)]
			\item Map file segments, in which \texttt{p\_vaddr $\equiv$ p\_offset$\pmod{\texttt{pg\_size}}$} for efficient mapping, to virtual memory segments
			\item Last \textbf{file page} of text segment may contain first page of data segment; last data page may contain file information not relevant to the running process to save space
			\item Segments’ addresses are adjusted to ensure each logical page in the address space has a single set of permissions as if each segment were complete and separate
			\item End of the data segment requires special handling, zero out, for uninitialized data; thus, (irrelevant) information not in the logical memory page must be zeroed out
			\item Executable file, shared object typically contains absolute code, PIC respectively
		\end{enumerate}
	\end{absnode}

    \createpin{exec1}{165,-147}
	\begin{absnode}[-1]{exec1}{90}{Virtual Address Space [System V ABI]}
		\begin{enumerate}[leftmargin=*,label=(\arabic*)]
			\item Only required to handle 48-bit addresses, therefore, conforming processes may only use addresses up to \texttt{0x7fffffffffff}
			\item Systems are permitted to use any $2^n$ page size in \texttt{[4KB, 64KB]}
			\item Conceptually, processes has full address space, but in practice process size is limited by system-reserved space, per-process limits, available combined physical memory and 2nd storage
			\item Dereferencing \texttt{null} pointer is erroneous, process should not expect \texttt{0x0} to be a valid address
			\item Applications may control their memory assignments
		\end{enumerate}
	\end{absnode}

		\item[\texttt{[\_]}] \texttt{[Information]} Linking\\
		$\hookrightarrow$ Symbol resolution (static) -- associating each symbol reference with exactly one symbol definition from the symbol tables of its input relocatable object files\\
		$\hookrightarrow$ Section relocation -- associating a memory location with each symbol definition, then modifying all references to those symbols so that they point to this memory location
		\item[\texttt{[\_]}] \texttt{[Information]} Static Library Creating\\
		$\hookrightarrow$ Collection of object files, usually comes in the form of \texttt{ar} archive files, augmented with a symbol table that accelerates the search for the needed files\\
		$\hookrightarrow$ Static library contributes its code to output
		\item[\texttt{[\_]}] \texttt{[Information]} Dynamic Library Creating\\
		$\hookrightarrow$ Dynamic libraries and executables usually have the same format, so only one loader is needed, and an executable can be used as a shared library\\
		$\hookrightarrow$ Shared library generally contributes symbol or dependency information to output
		\item[\texttt{[0]}] Why does 32-bit mode access GOT via \texttt{get\_pc\_thunk} instead of its absolute address?\\
